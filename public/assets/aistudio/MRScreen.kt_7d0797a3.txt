@Composable
fun MRScreen(
    currentModel: ModelItem?,
    clearTrigger: Int,
    calibration: MRCalibration,
    onCalibrationChanged: (MRCalibration) -> Unit,
    twoFingerMode: TwoFingerGestureMode = TwoFingerGestureMode.MOVE,
    onToggleTwoFingerMode: () -> Unit = {},
    onTripleTapFullScreen: () -> Unit = {},
    isPlacementLocked: Boolean = false,
    onToggleLock: (() -> Unit)? = null,
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    var hasCameraPermission by remember {
        mutableStateOf(
            ContextCompat.checkSelfPermission(context, Manifest.permission.CAMERA) == PackageManager.PERMISSION_GRANTED
        )
    }

    val permissionLauncher = rememberLauncherForActivityResult(
        contract = ActivityResultContracts.RequestPermission()
    ) { isGranted ->
        hasCameraPermission = isGranted
        if (isGranted) {
            DiagnosticsManager.log(LogCategory.AR, "Camera permission granted for MR dual camera")
        }
    }

    if (!hasCameraPermission) {
        Box(
            modifier = modifier
                .fillMaxSize()
                .background(Color.Black)
                .padding(24.dp)
                .testTag("mr_permission_screen"),
            contentAlignment = Alignment.Center
        ) {
            Card(
                colors = CardDefaults.cardColors(containerColor = Color(0xFF1E1E22)),
                shape = RoundedCornerShape(20.dp),
                elevation = CardDefaults.cardElevation(8.dp),
                modifier = Modifier.fillMaxWidth()
            ) {
                Column(
                    modifier = Modifier.padding(24.dp),
                    horizontalAlignment = Alignment.CenterHorizontally,
                    verticalArrangement = Arrangement.spacedBy(16.dp)
                ) {
                    Box(
                        modifier = Modifier
                            .size(60.dp)
                            .clip(CircleShape)
                            .background(Color(0xFF0084FF).copy(alpha = 0.2f)),
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Default.CameraAlt,
                            contentDescription = null,
                            tint = Color(0xFF0084FF),
                            modifier = Modifier.size(32.dp)
                        )
                    }

                    Text(
                        text = "Camera Access for Mixed Reality",
                        style = MaterialTheme.typography.titleMedium,
                        fontWeight = FontWeight.Bold,
                        color = Color.White,
                        textAlign = TextAlign.Center
                    )

                    Text(
                        text = "MR mode displays a Dual Camera + Dual 3D Model side-by-side stereoscopic view with rock-solid ground locking for Mixed Reality headsets.",
                        style = MaterialTheme.typography.bodyMedium,
                        textAlign = TextAlign.Center,
                        color = Color(0xFFAAAAAF)
                    )

                    Button(
                        onClick = { permissionLauncher.launch(Manifest.permission.CAMERA) },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF0084FF)),
                        shape = RoundedCornerShape(12.dp),
                        modifier = Modifier
                            .fillMaxWidth()
                            .testTag("grant_camera_permission_mr_button")
                    ) {
                        Text("Enable Camera")
                    }
                }
            }
        }
    } else {
        var mrRetryTrigger by remember { mutableIntStateOf(0) }
        var liveSessionError by remember { mutableStateOf<String?>(null) }
        var fallbackMounted by remember { mutableStateOf(false) }
        val (readiness, readinessMessage) = rememberARReadiness(context, mrRetryTrigger)
        val safePermissionHandler = remember(context) { SafeARPermissionHandler(context) }

        val needsFallback = readiness == ARReadiness.UNSUPPORTED_OR_FAILED || liveSessionError != null

        LaunchedEffect(needsFallback, mrRetryTrigger) {
            if (needsFallback) {
                fallbackMounted = false
                kotlinx.coroutines.delay(350L)
                fallbackMounted = true
            } else {
                fallbackMounted = false
            }
        }

        when {
            needsFallback -> {
                Box(
                    modifier = modifier
                        .fillMaxSize()
                        .background(Color.Black)
                ) {
                    if (fallbackMounted) {
                        ObjectScreen(
                            currentModel = currentModel,
                            resetTrigger = clearTrigger,
                            twoFingerMode = twoFingerMode,
                            onToggleTwoFingerMode = onToggleTwoFingerMode,
                            onTripleTapFullScreen = onTripleTapFullScreen,
                            modifier = Modifier.fillMaxSize()
                        )
                    } else {
                        Box(
                            modifier = Modifier.fillMaxSize(),
                            contentAlignment = Alignment.Center
                        ) {
                            CircularProgressIndicator(color = Color(0xFF0084FF))
                        }
                    }
                }
            }

            readiness == ARReadiness.CHECKING -> {
                Box(
                    modifier = modifier
                        .fillMaxSize()
                        .background(Color.Black),
                    contentAlignment = Alignment.Center
                ) {
                    CircularProgressIndicator(color = Color(0xFF0084FF))
                }
            }

            readiness == ARReadiness.NEEDS_INSTALL -> {
                Box(
                    modifier = modifier
                        .fillMaxSize()
                        .background(Color.Black)
                        .padding(24.dp),
                    contentAlignment = Alignment.Center
                ) {
                    Card(
                        colors = CardDefaults.cardColors(containerColor = Color(0xFF1E1E22)),
                        shape = RoundedCornerShape(20.dp),
                        modifier = Modifier.fillMaxWidth()
                    ) {
                        Column(
                            modifier = Modifier.padding(24.dp),
                            horizontalAlignment = Alignment.CenterHorizontally,
                            verticalArrangement = Arrangement.spacedBy(14.dp)
                        ) {
                            Text(
                                text = "ARCore Required for MR Tracking",
                                style = MaterialTheme.typography.titleMedium,
                                fontWeight = FontWeight.Bold,
                                color = Color.White
                            )
                            Text(
                                text = readinessMessage ?: "Install Google Play Services for AR to enable MR ground locking.",
                                style = MaterialTheme.typography.bodyMedium,
                                color = Color(0xFFAAAAAF),
                                textAlign = TextAlign.Center
                            )
                            Button(
                                onClick = {
                                    safePermissionHandler.requestARCoreInstall(true)
                                    mrRetryTrigger++
                                },
                                colors = ButtonDefaults.buttonColors(containerColor = Color(0xFF0084FF)),
                                modifier = Modifier.fillMaxWidth()
                            ) {
                                Text("Install / Retry ARCore")
                            }
                        }
                    }
                }
            }

            readiness == ARReadiness.READY -> {
                MRDualCameraExperience(
                    currentModel = currentModel,
                    clearTrigger = clearTrigger,
                    calibration = calibration,
                    twoFingerMode = twoFingerMode,
                    onToggleTwoFingerMode = onToggleTwoFingerMode,
                    onTripleTapFullScreen = onTripleTapFullScreen,
                    isPlacementLocked = isPlacementLocked,
                    onToggleLock = onToggleLock,
                    permissionHandler = safePermissionHandler,
                    onFatalSessionError = { errorMsg ->
                        liveSessionError = errorMsg
                    },
                    modifier = modifier
                )
            }
        }
    }
}

private class MRRuntimeHolder {
    var activePlacement: LockedPlacement? = null
    var latestScanState: LivePlaneScanState = LivePlaneScanState()
    var scanFrameCounter: Int = 0
    @Volatile var pendingTap: Pair<Float, Float>? = null
    @Volatile var ensurePlacedRequested: Boolean = false
    @Volatile var pendingReanchorAfterMove: Boolean = false
}

@Composable
private fun MRDualCameraExperience(
    currentModel: ModelItem?,
    clearTrigger: Int,
    calibration: MRCalibration,
    twoFingerMode: TwoFingerGestureMode,
    onToggleTwoFingerMode: () -> Unit,
    onTripleTapFullScreen: () -> Unit,
    isPlacementLocked: Boolean = false,
    onToggleLock: (() -> Unit)? = null,
    permissionHandler: SafeARPermissionHandler,
    onFatalSessionError: (String) -> Unit = {},
    modifier: Modifier = Modifier
) {
    val context = LocalContext.current
    val hostLifecycle = LocalLifecycleOwner.current.lifecycle

    val safeLifecycle = remember(hostLifecycle) {
        SafeARLifecycle(hostLifecycle) { throwable ->
            val msg = throwable.localizedMessage ?: throwable.javaClass.simpleName
            DiagnosticsManager.log(LogCategory.AR, "Caught MR lifecycle exception: $msg", isError = true)
            onFatalSessionError("MR Camera error: $msg")
        }
    }

    val cameraStabilityTracker = remember { ARCameraStabilityTracker() }
    val runtimeHolder = remember { MRRuntimeHolder() }
    val transformController = remember { LiveModelTransformController(initialScaleMultiplier = 1.0f) }

    var isCameraStreamReady by remember { mutableStateOf(false) }
    var isModelPlaced by remember { mutableStateOf(false) }
    var activeAnchor by remember { mutableStateOf<Anchor?>(null) }
    var liveScanState by remember { mutableStateOf(LivePlaneScanState()) }

    var leftEyeWidthPx by remember { mutableFloatStateOf(540f) }
    var leftEyeHeightPx by remember { mutableFloatStateOf(1080f) }

    val engine = rememberEngine()
    val modelLoader = rememberModelLoader(engine)
    val materialLoader = rememberMaterialLoader(engine)
    val environmentLoader = rememberEnvironmentLoader(engine)
    val studioEnvironment = rememberNaturalStudioEnvironment(engine, environmentLoader, intensity = 11_000f)
    val view = rememberARView(engine).apply {
        setShadowingEnabled(false)
        ambientOcclusionOptions = ambientOcclusionOptions.apply {
            enabled = false
        }
        setFrustumCullingEnabled(true)
    }
    val cameraStream = rememberARCameraStream(materialLoader).apply {
        isDepthOcclusionEnabled = false
        isPersonOcclusionEnabled = false
    }
    val cameraNode = rememberARCameraNode(engine).apply {
        near = 0.05f
        far = 50f
    }
    val mainLightNode = rememberMainLightNode(engine) {
        color = dev.romainguy.kotlin.math.Float4(1f, 1f, 1f, 1f)
        intensity = 90_000f
        lightDirection = Direction(-0.3f, -0.75f, -0.6f)
        isShadowCaster = false
    }
    val fillLightNode = rememberFillLightNode(engine) {
        color = dev.romainguy.kotlin.math.Float4(1f, 1f, 1f, 1f)
        intensity = 35_000f
        lightDirection = Direction(0.45f, -0.25f, 0.75f)
        isShadowCaster = false
    }

    // Pre-cache Filament 3D Model Instance in GPU memory for instant zero-flicker MR rendering
    val sceneAsset = rememberLoadedSceneAsset(
        modelLoader = modelLoader,
        modelItem = currentModel,
        instanceKey = currentModel?.id
    )
    LaunchedEffect(sceneAsset) {
        if (sceneAsset != null) {
            transformController.applyToNodes()
        }
    }

    val surfaceMirrorer = rememberSurfaceMirrorer()

    val activeRightSurfaceRef = remember { AtomicReference<Surface?>(null) }
    val retiredSurfaces = remember { java.util.Collections.synchronizedList(mutableListOf<Surface>()) }

    fun stopRightMirrorSafely() {
        val surf = activeRightSurfaceRef.getAndSet(null)
        if (surf != null) {
            runCatching { surfaceMirrorer.stopMirroring(surf) }
            retiredSurfaces.add(surf)
        }
    }

    fun detachAnchorOnly() {
        try {
            runtimeHolder.activePlacement?.anchor?.detach()
        } catch (_: Throwable) {}
    }

    DisposableEffect(surfaceMirrorer) {
        cameraStabilityTracker.startImuTracking(context)
        onDispose {
            cameraStabilityTracker.stopImuTracking()
            stopRightMirrorSafely()
            synchronized(retiredSurfaces) {
                for (s in retiredSurfaces) {
                    runCatching { s.release() }
                }
                retiredSurfaces.clear()
            }
        }
    }

    fun clearCurrentAnchor() {
        detachAnchorOnly()
        runtimeHolder.activePlacement = null
        activeAnchor = null
        transformController.isAnchored = false
        transformController.resetOffsets(defaultScale = 1.0f)
        isModelPlaced = false
    }

    fun applyNewPlacement(placement: LockedPlacement) {
        if (runtimeHolder.activePlacement?.anchor !== placement.anchor) {
            detachAnchorOnly()
        }
        runtimeHolder.activePlacement = placement
        activeAnchor = placement.anchor
        transformController.isAnchored = placement.anchor != null
        transformController.baseAnchorX = placement.worldPosition.x
        transformController.baseAnchorY = placement.worldPosition.y
        transformController.baseAnchorZ = placement.worldPosition.z
        transformController.baseAnchorYawDeg = placement.yawDegrees
        transformController.moveOffsetX = 0f
        transformController.moveOffsetY = 0f
        transformController.moveOffsetZ = 0f
        if (transformController.scaleMultiplier < 0.15f) {
            transformController.scaleMultiplier = 1.0f
        }
        transformController.applyToNodes()
        if (!isModelPlaced) {
            isModelPlaced = true
        }
    }

    LaunchedEffect(clearTrigger) {
        if (clearTrigger > 0) {
            clearCurrentAnchor()
            DiagnosticsManager.log(LogCategory.AR, "Reset MR stereo ground lock")
        }
    }

    LaunchedEffect(currentModel?.id) {
        if (currentModel != null && runtimeHolder.activePlacement == null) {
            runtimeHolder.ensurePlacedRequested = true
        }
    }

    // NOTE: Right eye is a hardware-mirrored copy (via SurfaceMirrorer) of the left-eye Filament View.
    // It operates in mono-passthrough mode. 2D offset provides horizontal adjustment for visual comfort
    // but does NOT generate true stereoscopic binocular parallax. Genuine 3D stereo requires dual Filament
    // Views with separated camera nodes.
    val ipdPixelOffset = ((calibration.ipdMm - 63f) * 1.5f).coerceIn(-30f, 30f)

    Box(
        modifier = modifier
            .fillMaxSize()
            .background(Color.Black)
            .testTag("mr_screen")
    ) {
        // ==================== DUAL CAMERA + DUAL 3D MODEL STEREOSCOPIC ROW ====================
        Row(
            modifier = Modifier.fillMaxSize()
        ) {
            // 1. LEFT EYE VIEWPORT (Primary ARSceneView: Live Camera + 3D Model + Floor Point Detection)
            Box(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .clip(RoundedCornerShape(0.dp))
                    .onSizeChanged { size ->
                        if (size.width > 0 && size.height > 0) {
                            leftEyeWidthPx = size.width.toFloat()
                            leftEyeHeightPx = size.height.toFloat()
                        }
                    }
                    .testTag("mr_left_eye")
            ) {
                ARSceneView(
                    modifier = Modifier.fillMaxSize(),
                    engine = engine,
                    view = view,
                    modelLoader = modelLoader,
                    materialLoader = materialLoader,
                    environmentLoader = environmentLoader,
                    cameraNode = cameraNode,
                    cameraStream = cameraStream,
                    mainLightNode = mainLightNode,
                    fillLightNode = fillLightNode,
                    permissionHandler = permissionHandler,
                    lifecycle = safeLifecycle,
                    surfaceMirrorer = surfaceMirrorer,
                    sessionCameraConfig = SafeARSessionConfigurator::selectOptimalCameraConfig,
                    sessionConfiguration = SafeARSessionConfigurator::configureSession,
                    focusMode = Config.FocusMode.AUTO,
                    planeRenderer = true,
                    sceneUnderstanding = SceneUnderstanding(
                        occlusion = false,
                        lighting = true,
                        physics = false,
                        planeVisualization = true
                    ),
                    planeFindingMode = Config.PlaneFindingMode.HORIZONTAL,
                    instantPlacementMode = Config.InstantPlacementMode.DISABLED,
                    depthMode = Config.DepthMode.DISABLED,
                    onSessionCreated = { session ->
                        view.setShadowingEnabled(false)
                        view.ambientOcclusionOptions = view.ambientOcclusionOptions.apply { enabled = false }
                        cameraStream.isDepthOcclusionEnabled = false
                        cameraStream.isPersonOcclusionEnabled = false
                        runCatching {
                            session.configure(session.config.apply {
                                SafeARSessionConfigurator.configureSession(session, this)
                            })
                        }
                        cameraStabilityTracker.onSessionStarted(context, session, leftEyeWidthPx, leftEyeHeightPx)
                    },
                    onSessionResumed = { session ->
                        view.setShadowingEnabled(false)
                        view.ambientOcclusionOptions = view.ambientOcclusionOptions.apply { enabled = false }
                        cameraStream.isDepthOcclusionEnabled = false
                        cameraStream.isPersonOcclusionEnabled = false
                        runCatching {
                            session.configure(session.config.apply {
                                SafeARSessionConfigurator.configureSession(session, this)
                            })
                        }
                        cameraStabilityTracker.onSessionStarted(context, session, leftEyeWidthPx, leftEyeHeightPx)
                    },
                    onSessionFailed = { exception ->
                        val msg = exception.localizedMessage ?: exception.javaClass.simpleName
                        DiagnosticsManager.log(LogCategory.AR, "MR session failed: $msg", isError = true)
                        onFatalSessionError("MR session unavailable ($msg)")
                    },
                    onSessionUpdated = { session, frame ->
                        DiagnosticsManager.onFrameRendered()

                        cameraStabilityTracker.syncFrame(
                            context = context,
                            session = session,
                            frame = frame,
                            cameraNode = cameraNode,
                            viewWidthPx = leftEyeWidthPx,
                            viewHeightPx = leftEyeHeightPx
                        )

                        if (!isCameraStreamReady && cameraStabilityTracker.validCameraFrames >= 4) {
                            isCameraStreamReady = true
                        }

                        val camera = frame.camera
                        val state = camera.trackingState
                        val effectiveCamPose = cameraStabilityTracker.getEffectiveCameraPose(frame)
                        val currentLock = runtimeHolder.activePlacement
                        val tap = runtimeHolder.pendingTap

                        // Ensure the rendering loop doesn't override or update the node's transformation matrix manually while Ground Lock is locked
                        if (isPlacementLocked && isModelPlaced) {
                            runtimeHolder.pendingTap = null
                            runtimeHolder.ensurePlacedRequested = false
                            runtimeHolder.pendingReanchorAfterMove = false
                            return@ARSceneView
                        }

                        val needsScan = currentLock == null ||
                            !currentLock.isGroundedOnPlane ||
                            currentLock.anchor == null ||
                            tap != null ||
                            runtimeHolder.ensurePlacedRequested

                        if (needsScan) {
                            runtimeHolder.scanFrameCounter++
                            if (tap != null || runtimeHolder.ensurePlacedRequested || runtimeHolder.scanFrameCounter % 3 == 0) {
                                val scan = GroundLockController.scanLivePlanesAndReticle(
                                    session = session,
                                    frame = frame,
                                    viewWidthPx = leftEyeWidthPx,
                                    viewHeightPx = leftEyeHeightPx,
                                    effectiveCameraPose = effectiveCamPose
                                )
                                runtimeHolder.latestScanState = scan
                                if (currentLock == null) {
                                    liveScanState = scan
                                }
                            }
                        }

                        // Re-anchor only when user manually unlocks and drags the model
                        if (runtimeHolder.pendingReanchorAfterMove && !isPlacementLocked) {
                            runtimeHolder.pendingReanchorAfterMove = false
                            val curPlacement = runtimeHolder.activePlacement
                            if (curPlacement != null && state == TrackingState.TRACKING) {
                                val reanchored = GroundLockController.reanchorMovedPlacement(
                                    session = session,
                                    frame = frame,
                                    current = curPlacement,
                                    newWorldX = transformController.baseAnchorX,
                                    newWorldZ = transformController.baseAnchorZ,
                                    scanState = runtimeHolder.latestScanState
                                )
                                if (reanchored.anchor != null) {
                                    applyNewPlacement(reanchored)
                                }
                            }
                        }

                        // Real-World Surface Scanning: ONLY place on a real tracked horizontal floor or Depth API surface
                        if (runtimeHolder.ensurePlacedRequested && (state == TrackingState.TRACKING || frame.timestamp > 0L)) {
                            if (runtimeHolder.activePlacement == null) {
                                val allPlanes = runCatching { session.getAllTrackables(Plane::class.java) }.getOrDefault(emptyList())
                                val trackingPlanes = allPlanes.filter {
                                    it.trackingState == TrackingState.TRACKING &&
                                        it.subsumedBy == null &&
                                        it.type == Plane.Type.HORIZONTAL_UPWARD_FACING
                                }
                                val newPlacement = GroundLockController.createCenterGroundedPlacement(
                                    session = session,
                                    frame = frame,
                                    trackingPlanes = trackingPlanes,
                                    viewWidthPx = leftEyeWidthPx,
                                    viewHeightPx = leftEyeHeightPx,
                                    effectiveCameraPose = effectiveCamPose
                                )
                                if (newPlacement != null && newPlacement.anchor != null && newPlacement.anchor.trackingState != TrackingState.STOPPED) {
                                    runtimeHolder.ensurePlacedRequested = false
                                    applyNewPlacement(newPlacement)
                                    DiagnosticsManager.log(LogCategory.AR, "MR placed model on real ground anchor")
                                }
                            } else {
                                runtimeHolder.ensurePlacedRequested = false
                            }
                        }

                        // The ARCore anchor is maintained natively by ARCore SLAM with zero drift.
                        // Only clear if the anchor was permanently STOPPED by the ARCore session.
                        val activeLock = runtimeHolder.activePlacement
                        if (activeLock != null && activeLock.anchor?.trackingState == TrackingState.STOPPED) {
                            DiagnosticsManager.log(LogCategory.AR, "MR active anchor stopped; clearing")
                            runCatching { activeLock.anchor.detach() }
                            activeAnchor = null
                            runtimeHolder.activePlacement = null
                            isModelPlaced = false
                            transformController.isAnchored = false
                        }

                        // 1 Tap: Point on Floor in MR (places model solidly to tapped point on the floor)
                        if (tap != null && !isPlacementLocked) {
                            val tapX = tap.first
                            val tapY = tap.second
                            runtimeHolder.pendingTap = null
                            try {
                                val tapPlacement = GroundLockController.createGroundedPlacementFromTap(
                                    session = session,
                                    frame = frame,
                                    screenX = tapX,
                                    screenY = tapY,
                                    viewWidthPx = leftEyeWidthPx,
                                    viewHeightPx = leftEyeHeightPx,
                                    scanState = runtimeHolder.latestScanState,
                                    effectiveCameraPose = effectiveCamPose,
                                    existingFloorY = runtimeHolder.activePlacement?.worldPosition?.y
                                )
                                if (tapPlacement != null && tapPlacement.anchor != null) {
                                    applyNewPlacement(tapPlacement)
                                }
                            } catch (_: Exception) {}
                        }
                    }
                ) {
                    if (!isModelPlaced) {
                        RenderDetectedPlanesAndTargetReticle(scanState = liveScanState)
                    }

                    if (currentModel != null && isModelPlaced && sceneAsset != null) {
                        val currentAnchor = activeAnchor
                        if (currentAnchor != null) {
                            transformController.isAnchored = true
                            AnchorNode(
                                anchor = currentAnchor,
                                updateAnchorPose = !isPlacementLocked, // Freeze the pose/anchor transformation when locked
                                visibleTrackingStates = setOf(TrackingState.TRACKING, TrackingState.PAUSED),
                                apply = {
                                    this.isEditable = false
                                    this.isPositionEditable = false
                                    this.isRotationEditable = false
                                    this.isScaleEditable = false
                                    this.isSmoothTransformEnabled = false
                                    this.smoothTransformSpeed = 0f
                                    this.visibleCameraTrackingStates = setOf(TrackingState.TRACKING, TrackingState.PAUSED)
                                    this.visibleTrackingStates = setOf(TrackingState.TRACKING, TrackingState.PAUSED)
                                }
                            ) {
                                Node(
                                    position = Position(0f, 0f, 0f),
                                    rotation = Rotation(
                                        x = 0f,
                                        y = transformController.yawDeg,
                                        z = 0f
                                    ),
                                    scale = Scale(transformController.scaleMultiplier),
                                    isVisible = true,
                                    apply = {
                                        transformController.isAnchored = true
                                        transformController.outerNode = this
                                    }
                                ) {
                                    RenderMuseumOrModelContent(
                                        currentModel = currentModel,
                                        sceneAsset = sceneAsset,
                                        scaleMultiplier = transformController.scaleMultiplier,
                                        yRotationDeg = transformController.yawDeg,
                                        xRotationDeg = 0f,
                                        autoAnimate = false,
                                        transformController = transformController
                                    )
                                }
                            }
                        }
                    }
                }
            }

            // 2. OPTICAL CENTER DIVIDER
            if (calibration.showCenterDivider) {
                Box(
                    modifier = Modifier
                        .width(4.dp)
                        .fillMaxHeight()
                        .background(Color.Black)
                )
            }

            // 3. RIGHT EYE VIEWPORT (GPU-Mirrored Live AR Camera + 3D Model via SurfaceMirrorer)
            Box(
                modifier = Modifier
                    .weight(1f)
                    .fillMaxHeight()
                    .clip(RoundedCornerShape(0.dp))
                    .testTag("mr_right_eye")
            ) {
                AndroidView(
                    modifier = Modifier
                        .fillMaxSize()
                        .graphicsLayer {
                            translationX = -ipdPixelOffset
                        },
                    factory = { ctx ->
                        TextureView(ctx).apply {
                            isOpaque = true
                            surfaceTextureListener = object : TextureView.SurfaceTextureListener {
                                override fun onSurfaceTextureAvailable(
                                    st: SurfaceTexture,
                                    width: Int,
                                    height: Int
                                ) {
                                    stopRightMirrorSafely()
                                    val surf = Surface(st)
                                    activeRightSurfaceRef.set(surf)
                                    runCatching {
                                        surfaceMirrorer.startMirroring(
                                            surf,
                                            width = width.coerceAtLeast(1),
                                            height = height.coerceAtLeast(1)
                                        )
                                        DiagnosticsManager.log(
                                            LogCategory.RENDER,
                                            "Right eye MR camera + 3D mirror started (${width}x$height)"
                                        )
                                    }
                                }

                                override fun onSurfaceTextureSizeChanged(
                                    st: SurfaceTexture,
                                    width: Int,
                                    height: Int
                                ) {
                                    val surf = activeRightSurfaceRef.get()
                                    if (surf != null) {
                                        runCatching { surfaceMirrorer.stopMirroring(surf) }
                                        runCatching {
                                            surfaceMirrorer.startMirroring(
                                                surf,
                                                width = width.coerceAtLeast(1),
                                                height = height.coerceAtLeast(1)
                                            )
                                        }
                                    }
                                }

                                override fun onSurfaceTextureDestroyed(st: SurfaceTexture): Boolean {
                                    stopRightMirrorSafely()
                                    return true
                                }

                                override fun onSurfaceTextureUpdated(st: SurfaceTexture) {}
                            }
                        }
                    },
                    onRelease = {
                        stopRightMirrorSafely()
                    }
                )
            }
        }

        // Declared AFTER ARSceneView and TextureView so LIFO disposal stops mirroring BEFORE Engine teardown
        DisposableEffect(Unit) {
            onDispose {
                cameraStabilityTracker.stopImuTracking()
                stopRightMirrorSafely()
                detachAnchorOnly()
            }
        }

        // Prevent the 1-frame yellow/green uninitialized external OES camera texture flash on startup
        if (!isCameraStreamReady) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black),
                contentAlignment = Alignment.Center
            ) {
                CircularProgressIndicator(color = Color(0xFF0084FF))
            }
        }

        // Zero-recomposition Touch Gesture Layer across both eyes:
        // - 1 tap: point on floor
        // - 1 finger: rotation (360° upright spin on floor)
        // - 2 fingers: moving on floor + pinch scale (*10000 in and out)
        // Primary Interaction Viewport Touch Gesture Layer (constrained to Left Eye AR Viewport):
        // Prevents multi-touch gestures spanning across both eyes from triggering erratic jumps.
        Box(
            modifier = Modifier
                .fillMaxHeight()
                .fillMaxWidth(0.5f)
                .modelTouchGestures(
                    onSingleTapFloorPoint = { tapX, tapY ->
                        if (isPlacementLocked && isModelPlaced) {
                            DiagnosticsManager.log(LogCategory.AR, "MR tap ignored: placement is locked")
                        } else {
                            runtimeHolder.pendingTap = Pair(tapX, tapY)
                        }
                    },
                    onDoubleTapSwitchMode = onToggleTwoFingerMode,
                    onTripleTapFullScreen = onTripleTapFullScreen,
                    onOneFingerRotate = { dxPx, _ ->
                        if (isPlacementLocked) return@modelTouchGestures
                        if (runtimeHolder.activePlacement == null) {
                            runtimeHolder.ensurePlacedRequested = true
                        }
                        val totalWidth = leftEyeWidthPx.coerceAtLeast(1f)
                        val yawDelta = (dxPx / totalWidth) * 360f * 1.6f
                        transformController.yawDeg = ((transformController.yawDeg + yawDelta) % 360f + 360f) % 360f
                        transformController.pitchDeg = 0f
                        transformController.applyToNodes()
                    },
                    onTwoFingerGesture = { spanRatio, centroidDx, centroidDy, _ ->
                        if (isPlacementLocked) return@modelTouchGestures
                        val delta = spanRatio - 1f
                        val factor = 1f + (delta * 0.70f).coerceIn(-0.25f, 0.25f)
                        if (abs(delta) > 0.002f) {
                            transformController.scaleMultiplier = (transformController.scaleMultiplier * factor)
                                .coerceIn(MIN_INFINITE_SCALE, MAX_INFINITE_SCALE)
                        }
                        if (twoFingerMode == TwoFingerGestureMode.MOVE && !isPlacementLocked) {
                            val (fwdX, fwdZ) = cameraStabilityTracker.lastHorizontalForward
                            val rightX = -fwdZ
                            val rightZ = fwdX
                            val dist = 1.5f
                            val totalWidth = leftEyeWidthPx.coerceAtLeast(1f)
                            val totalHeight = leftEyeHeightPx.coerceAtLeast(1f)
                            val moveRightM = (centroidDx / totalWidth) * dist * 1.5f
                            val moveForwardM = (-centroidDy / totalHeight) * dist * 1.5f
                            transformController.baseAnchorX += rightX * moveRightM + fwdX * moveForwardM
                            transformController.baseAnchorZ += rightZ * moveRightM + fwdZ * moveForwardM
                            runtimeHolder.pendingReanchorAfterMove = true
                        }
                        transformController.applyToNodes()
                    },
                    onTwoFingerGestureEnd = {
                        // Coordinates updated while dragging; re-anchoring to fixed AR anchor occurs in onSessionUpdated
                    }
                )
        )

        // Real-World Surface Scanning guidance and Anchor tracking HUD for MR
        RealSurfaceScanHUD(
            isModelPlaced = isModelPlaced,
            scanState = liveScanState,
            isPlacementLocked = isPlacementLocked,
            onToggleLock = onToggleLock
        )
    }
}
